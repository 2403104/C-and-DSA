void solve()
{
    int n, k;
    cin >> n >> k;
    vi arr(n);
    input(arr);
    vi b = arr;
    sort(all(b));
    int val = b[k - 1];
    b.clear();
    deque<int> d;
    for (int i = 0; i < n; i++)
    {
        if (arr[i] <= val)
            d.push_back(arr[i]);
    }
    ll cnt = d.size() - k + 1;
    while (d.size() > 1)
    {
        if (d.front() < val && d.back() < val)
        {
            if (d.front() != d.back())
            {
                no;
                return;
            }
            d.pop_front();
            d.pop_back();
        }
        else if (d.front() == val && d.back() == val)
        {
            d.pop_front();
            d.pop_back();
        }
        else
        {
            if (!cnt)
            {
                no;
                return;
            }
            else
            {
                if (d.back() == val)
                    d.pop_back();
                else
                    d.pop_front();
                cnt--;
            }
        }
    }
    yes;
}
